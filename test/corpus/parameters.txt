==================
parameter tag with single param
==================
<%- | $x | -%>
hello <%= $x %>
---

(template
  (parameter_directive
    (parameter_list
      (parameter
        name: (parameter_name))))
  (content)
  (output_directive
    (code
      (variable))))

==================
parameter tag with default value
==================
<%- | $x, $y = 'default' | -%>
body
---

(template
  (parameter_directive
    (parameter_list
      (parameter
        name: (parameter_name))
      (parameter
        name: (parameter_name)
        default: (parameter_default))))
  (content))
