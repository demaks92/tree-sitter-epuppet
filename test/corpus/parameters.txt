==================
Parameter tag with single param
==================
<%- | $x | -%>
hello <%= $x %>
---

(template
  (parameter_directive
    (parameter_list
      (parameter
        name: (parameter_name))))
  (content)
  (output_directive
    (code
      (variable))))

---

    (parameter_list

        type: (parameter_type
          type: (type_name))

      (parameter
        type: (parameter_type
          type: (type_name))
        name: (parameter_name))
      (parameter
        type: (parameter_type
          type: (array_type)
          arguments: (array_type_arguments
            value: (type_argument
              (parameter_type
                type: (type_name)))))
        name: (parameter_name))))
  (content))
